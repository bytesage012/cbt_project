=== app/page.tsx ===
import React from "react";
import Header from "@/components/Header";
import SubjectCard from "@/components/SubjectCard";
import AddSubjectForm from "@/components/AddSubjectForm";

type Subject = {
  id: string;
  name: string;
  description: string;
  _count: { courses: number };
};

export const dynamic = "force-dynamic";

export const metadata = {
  title: "CBT Prep Hub — Choose a Subject",
  description: "Select a subject to browse courses and start practising exam questions.",
};

export default async function HomePage() {
  const res = await fetch(`${process.env.NEXT_PUBLIC_BASE_URL}/api/subjects`);
  const subjects: Subject[] = await res.json();

  return (
    <>
      <Header />
      <main className="flex-1 bg-navy px-6 py-12 md:px-10 lg:px-16">
        {/* Hero */}
        <section className="text-center mb-14 max-w-2xl mx-auto">
          <p className="text-gold text-xs font-semibold uppercase tracking-widest mb-3">
            Your Exam Platform
          </p>
          <h1 className="text-5xl font-bold tracking-tight text-offwhite mb-4 leading-tight">
            Welcome to{" "}
            <span className="text-gold">CBT Prep Hub</span>
          </h1>
          <p className="text-muted text-lg leading-relaxed">
            Choose a subject below, browse its courses, and start practising
            with real exam-style questions.
          </p>
        </section>

        {/* Subjects grid */}
        {subjects.length === 0 ? (
          <div className="text-center py-20">
            <p className="text-5xl mb-4">📚</p>
            <p className="text-offwhite font-semibold text-lg mb-1">No subjects yet</p>
            <p className="text-muted text-sm">Add your first subject to get started.</p>
          </div>
        ) : (
          <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-5 max-w-6xl mx-auto">
            {subjects.map((s) => (
              <SubjectCard
                key={s.id}
                id={s.id}
                name={s.name}
                description={s.description}
                _count={s._count}
              />
            ))}
          </div>
        )}

        {/* Add Subject */}
        <AddSubjectForm />
      </main>
    </>
  );
}
