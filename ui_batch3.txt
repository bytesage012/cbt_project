"use client";

import React, { useState } from "react";
import Link from "next/link";
import { usePathname } from "next/navigation";

const MoonIcon = () => (
  <svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
    <path d="M21 12.79A9 9 0 1 1 11.21 3 7 7 0 0 0 21 12.79z"/>
  </svg>
);

const SunIcon = () => (
  <svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
    <circle cx="12" cy="12" r="5"/>
    <line x1="12" y1="1" x2="12" y2="3"/>
    <line x1="12" y1="21" x2="12" y2="23"/>
    <line x1="4.22" y1="4.22" x2="5.64" y2="5.64"/>
    <line x1="18.36" y1="18.36" x2="19.78" y2="19.78"/>
    <line x1="1" y1="12" x2="3" y2="12"/>
    <line x1="21" y1="12" x2="23" y2="12"/>
    <line x1="4.22" y1="19.78" x2="5.64" y2="18.36"/>
    <line x1="18.36" y1="5.64" x2="19.78" y2="4.22"/>
  </svg>
);

const Header: React.FC = () => {
  const [dark, setDark] = useState(false);
  const pathname = usePathname();

  const toggleDark = () => {
    const next = !dark;
    setDark(next);
    document.documentElement.classList.toggle("dark", next);
  };

  const navLink = (href: string, label: string) => {
    const isActive = pathname === href;
    return (
      <Link
        href={href}
        className={`text-sm font-medium transition-colors duration-150 ${
          isActive
            ? "text-gold"
            : "text-offwhite/70 hover:text-offwhite"
        }`}
      >
        {label}
      </Link>
    );
  };

  return (
    <header className="sticky top-0 z-50 h-16 flex items-center px-8 bg-navy-surface border-b border-navy-border">
      {/* Logo */}
      <Link href="/" className="flex items-center gap-2 mr-10 select-none">
        <span className="text-2xl" aria-hidden="true">🎓</span>
        <span className="text-lg font-bold tracking-tight text-offwhite">
          CBT <span className="text-gold">Prep Hub</span>
        </span>
      </Link>

      {/* Nav */}
      <nav className="flex items-center gap-6 flex-1">
        {navLink("/", "Home")}
        {navLink("/config", "Settings")}
      </nav>

      {/* Dark mode toggle */}
      <button
        id="dark-mode-toggle"
        onClick={toggleDark}
        aria-label="Toggle dark mode"
        className="p-2 rounded-lg border border-navy-border text-muted hover:text-gold hover:border-gold/50 transition-all duration-150"
      >
        {dark ? <SunIcon /> : <MoonIcon />}
      </button>
    </header>
  );
};

export default Header;

"use client";
import React from "react";
import type { Question } from "@/lib/types";
import { optionLabels } from "@/lib/types";

type ExamQuestionProps = {
  question: Question;
  questionNumber: number;
  total: number;
  selectedAnswer: string | null;
  onAnswer: (answer: string) => void;
};

export default function ExamQuestion({
  question,
  questionNumber,
  total,
  selectedAnswer,
  onAnswer,
}: ExamQuestionProps) {
  const options = JSON.parse(question.options) as string[];
  const progress = Math.round((questionNumber / total) * 100);

  return (
    <div className="w-full max-w-3xl mx-auto animate-slide-up">
      {/* Progress bar */}
      <div className="mb-6">
        <div className="flex justify-between items-center mb-2">
          <span className="text-xs text-muted font-medium uppercase tracking-wider">
            Question {questionNumber} of {total}
          </span>
          <span className="text-xs text-gold font-semibold">{progress}%</span>
        </div>
        <div className="h-1 bg-navy-border rounded-full overflow-hidden">
          <div
            className="h-full bg-gold rounded-full transition-all duration-500"
            style={{ width: `${progress}%` }}
          />
        </div>
      </div>

      {/* Question prompt */}
      <p className="text-2xl font-bold text-offwhite leading-snug mb-8">
        {question.prompt}
      </p>

      {/* Options grid — 2×2 on desktop, stacked on mobile */}
      <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
        {options.map((opt, idx) => {
          const label = optionLabels[idx];
          const isSelected = selectedAnswer === label;

          return (
            <button
              key={label}
              onClick={() => onAnswer(label)}
              className={`flex items-center gap-4 p-4 rounded-xl border text-left transition-all duration-150 ${
                isSelected
                  ? "bg-gold/10 border-gold shadow-[0_0_0_1px_rgba(201,168,76,0.4)]"
                  : "bg-navy-surface border-navy-border hover:border-gold/50 hover:bg-navy-deep/60"
              }`}
            >
              {/* Letter badge */}
              <span
                className={`w-8 h-8 flex-shrink-0 flex items-center justify-center rounded-lg text-sm font-bold transition-all duration-150 ${
                  isSelected
                    ? "bg-gold text-navy"
                    : "bg-navy-deep text-gold border border-navy-border"
                }`}
              >
                {label}
              </span>
              <span className={`text-sm font-medium leading-snug ${isSelected ? "text-offwhite" : "text-offwhite/80"}`}>
                {opt}
              </span>
            </button>
          );
        })}
      </div>
    </div>
  );
}
