"use client";
import React, { useEffect, useState } from "react";

type ExamTimerProps = {
  durationSeconds: number;
  onTimeUp: () => void;
};

const ClockIcon = () => (
  <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round">
    <circle cx="12" cy="12" r="10"/>
    <polyline points="12 6 12 12 16 14"/>
  </svg>
);

export default function ExamTimer({ durationSeconds, onTimeUp }: ExamTimerProps) {
  const [secondsLeft, setSecondsLeft] = useState(durationSeconds);

  useEffect(() => {
    if (secondsLeft <= 0) { onTimeUp(); return; }
    const id = setInterval(() => setSecondsLeft((s) => s - 1), 1000);
    return () => clearInterval(id);
  }, [secondsLeft, onTimeUp]);

  const minutes = Math.floor(secondsLeft / 60);
  const seconds = secondsLeft % 60;
  const timeStr = `${minutes.toString().padStart(2, "0")}:${seconds.toString().padStart(2, "0")}`;

  let colorClass = "text-offwhite border-navy-border";
  let pulse = "";
  if (secondsLeft <= 30) {
    colorClass = "text-red-400 border-red-400";
    pulse = "animate-pulse";
  } else if (secondsLeft <= 60) {
    colorClass = "text-gold border-gold";
  }

  return (
    <div
      className={`flex items-center gap-2 bg-navy-surface rounded-full px-4 py-2 border text-sm font-mono font-semibold ${colorClass} ${pulse} transition-colors duration-300`}
    >
      <ClockIcon />
      {timeStr}
    </div>
  );
}

  total: number;

  const percent = Math.round((correct / total) * 100);
  const passed  = percent >= 50;

  const mins   = Math.floor(timeTaken / 60);

  const timeStr = `${mins.toString().padStart(2, "0")}:${secs.toString().padStart(2, "0")}`;

  return (
    <div className="min-h-screen bg-navy flex items-center justify-center p-6">
      <div className="w-full max-w-lg animate-slide-up">
        {/* Score header card */}
        <div className="bg-navy-surface border border-navy-border rounded-2xl p-8 text-center mb-4">
          {/* Big percentage */}
          <p className={`text-7xl font-bold mb-3 ${passed ? "text-gold" : "text-red-400"}`}>
            {percent}<span className="text-3xl">%</span>
          </p>

          {/* Pass/Fail badge */}
          <span className={`badge text-sm px-4 py-1.5 mb-6 inline-block ${passed ? "badge-green" : "badge-red"}`}>
            {passed ? "✓ PASSED" : "✗ FAILED"}
          </span>

          {/* Stats row */}
          <div className="grid grid-cols-3 gap-3 mt-6">
            {[
              { label: "Total",   value: String(total)   },
              { label: "Correct", value: String(correct) },
              { label: "Time",    value: timeStr         },
            ].map(({ label, value }) => (
              <div key={label} className="bg-navy-deep border border-navy-border rounded-xl p-3">
                <p className="text-xl font-bold text-offwhite">{value}</p>
                <p className="text-xs text-muted mt-0.5">{label}</p>
              </div>
            ))}
          </div>
        </div>

        {/* Details toggle */}
        <button
          className="btn-secondary w-full mb-3"
          onClick={() => setExpanded(!expanded)}
        >
          {expanded ? "Hide Details ↑" : "Show Details ↓"}
        </button>

        {/* Result rows */}
        {expanded && (
          <div className="bg-navy-surface border border-navy-border rounded-2xl overflow-hidden mb-4">
            <ul className="divide-y divide-navy-border">

import React, { useState } from "react";
import type { Question } from "@/lib/types";

type QuestionFormProps = {
  courseId: string;
  question?: Question;
  onSuccess: () => void;
  onCancel: () => void;
};

const OPTION_LABELS = ["A", "B", "C", "D"];

export default function QuestionForm({ courseId, question, onSuccess, onCancel }: QuestionFormProps) {
  const [prompt, setPrompt]       = useState(question?.prompt ?? "");
  const [options, setOptions]     = useState<string[]>(
    question ? JSON.parse(question.options) : ["", "", "", ""]
  );
  const [answer, setAnswer]       = useState(question?.answer ?? "A");
  const [difficulty, setDifficulty] = useState(question?.difficulty ?? "easy");
  const [loading, setLoading]     = useState(false);

  const handleOptionChange = (idx: number, value: string) => {
    const copy = [...options];
    copy[idx] = value;
    setOptions(copy);
  };

  const submit = async (e: React.FormEvent) => {
    e.preventDefault();
    setLoading(true);
    const payload = {
      prompt,
      options: JSON.stringify(options),
      answer,
      difficulty,
      courseId: Number(courseId),
    };
    const url    = question ? `/api/questions/${question.id}` : "/api/questions";
    const method = question ? "PATCH" : "POST";
    const res    = await fetch(url, {
      method,
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(payload),
    });
    setLoading(false);
    if (res.ok) onSuccess();
  };

  return (
    <form className="space-y-5" onSubmit={submit}>
      <h2 className="text-lg font-bold text-offwhite">
        {question ? "Edit Question" : "New Question"}
      </h2>

      {/* Prompt */}
      <div>
        <label className="block text-xs text-muted mb-1.5">Question Prompt</label>
        <textarea
          className="input resize-none"
          rows={3}
          placeholder="Enter the question text…"
          value={prompt}
          onChange={(e) => setPrompt(e.target.value)}
          required
        />
      </div>

      {/* Options */}
      <div className="space-y-2.5">
        <label className="block text-xs text-muted mb-1">Answer Options</label>
        {OPTION_LABELS.map((label, i) => (
          <div key={label} className="flex items-center gap-3">
            <span className="w-7 h-7 flex-shrink-0 flex items-center justify-center rounded-md bg-gold/20 text-gold text-xs font-bold">
              {label}
            </span>
            <input
              className="input"
              placeholder={`Option ${label}`}
              value={options[i]}
              onChange={(e) => handleOptionChange(i, e.target.value)}
              required
            />
          </div>
        ))}
      </div>

      {/* Correct answer + Difficulty row */}
      <div className="grid grid-cols-2 gap-4">
        <div>
          <label className="block text-xs text-muted mb-1.5">Correct Answer</label>
          <select
            className="input"
            value={answer}
            onChange={(e) => setAnswer(e.target.value)}
          >
            {OPTION_LABELS.map((l) => (
              <option key={l} value={l}>{l}</option>
            ))}
          </select>
        </div>
        <div>
          <label className="block text-xs text-muted mb-1.5">Difficulty</label>
          <select
            className="input"
            value={difficulty}
            onChange={(e) => setDifficulty(e.target.value)}
          >
            <option value="easy">Easy</option>
            <option value="medium">Medium</option>
            <option value="hard">Hard</option>
          </select>
        </div>
      </div>

      {/* Actions */}
      <div className="flex flex-col gap-2 pt-1">
        <button type="submit" className="btn-primary w-full" disabled={loading}>
          {loading ? "Saving…" : question ? "Update Question" : "Add Question"}
        </button>
        <button type="button" className="btn-secondary w-full" onClick={onCancel}>
          Cancel
        </button>
      </div>
    </form>
  );
}

import React, { useState } from "react";
import type { Question } from "@/lib/types";

type QuestionTableProps = {
  questions: Question[];
  onDelete: (id: string) => void;
  onEdit: (q: Question) => void;
};

const difficultyBadge = (d: string) => {
  if (d === "hard")   return "badge-red";
  if (d === "medium") return "badge-gold";
  return "badge-green";
};

export default function QuestionTable({ questions, onDelete, onEdit }: QuestionTableProps) {
  const [page, setPage] = useState(0);
  const pageSize = 10;
  const totalPages = Math.max(1, Math.ceil(questions.length / pageSize));
  const visible = questions.slice(page * pageSize, (page + 1) * pageSize);

  return (
    <div className="space-y-4 animate-fade-in">
      {/* Table wrapper */}
      <div className="bg-navy-surface border border-navy-border rounded-2xl overflow-hidden">
        <table className="min-w-full text-sm">
          {/* Header */}
          <thead>
            <tr className="bg-navy-deep border-b border-navy-border">
              <th className="px-5 py-3 text-left text-xs text-muted uppercase tracking-wider w-10">#</th>
              <th className="px-5 py-3 text-left text-xs text-muted uppercase tracking-wider">Prompt</th>
              <th className="px-5 py-3 text-center text-xs text-muted uppercase tracking-wider w-20">Answer</th>
              <th className="px-5 py-3 text-center text-xs text-muted uppercase tracking-wider w-28">Difficulty</th>
              <th className="px-5 py-3 text-center text-xs text-muted uppercase tracking-wider w-28">Actions</th>
            </tr>
          </thead>

          {/* Body */}
          <tbody>
            {visible.length === 0 ? (
              <tr>
                <td colSpan={5} className="px-5 py-10 text-center text-muted">
                  No questions yet. Add one to get started.
                </td>
              </tr>
            ) : (
              visible.map((q, idx) => (
                <tr
                  key={q.id}
                  className="border-b border-navy-border last:border-0 hover:bg-navy-deep/50 transition-colors duration-100"
                >
                  <td className="px-5 py-3 text-muted text-center">
                    {page * pageSize + idx + 1}
                  </td>
                  <td className="px-5 py-3 text-offwhite max-w-xs truncate" title={q.prompt}>
                    {q.prompt.length > 70 ? q.prompt.slice(0, 67) + "…" : q.prompt}
                  </td>
                  <td className="px-5 py-3 text-center">
                    <span className="badge">{q.answer}</span>
                  </td>
                  <td className="px-5 py-3 text-center">
                    <span className={`badge ${difficultyBadge(q.difficulty ?? "easy")}`}>
                      {q.difficulty ?? "easy"}
                    </span>
                  </td>
                  <td className="px-5 py-3 text-center">
                    <div className="flex items-center justify-center gap-2">
                      <button
                        className="text-gold hover:bg-gold/10 text-xs font-semibold px-2 py-1 rounded transition-colors duration-100"
                        onClick={() => onEdit(q)}
                      >
                        Edit
                      </button>
                      <button
                        className="text-red-400 hover:bg-red-500/10 text-xs font-semibold px-2 py-1 rounded transition-colors duration-100"
                        onClick={() => onDelete(String(q.id))}
                      >
                        Delete
                      </button>
                    </div>
                  </td>
                </tr>
              ))
            )}
          </tbody>
        </table>
      </div>

      {/* Pagination */}
      <div className="flex items-center justify-between px-1">
        <button
          disabled={page === 0}
          className="btn-secondary text-xs py-1.5 px-3 disabled:opacity-30 disabled:cursor-not-allowed"
          onClick={() => setPage((p) => Math.max(p - 1, 0))}
        >
          ← Prev
        </button>
        <span className="text-xs text-muted">
          Page {page + 1} of {totalPages}
        </span>
        <button
          disabled={page >= totalPages - 1}
          className="btn-secondary text-xs py-1.5 px-3 disabled:opacity-30 disabled:cursor-not-allowed"
          onClick={() => setPage((p) => Math.min(p + 1, totalPages - 1))}
        >
          Next →
        </button>
      </div>
    </div>
  );
}
